  <script>
    const fmt = n => {
      const num = parseFloat(n);
      return num % 1 === 0 ? num.toString() : num.toFixed(2).replace(/\.?0+$/, '');
    };

    const buatTabel = (sediaan, frekMap) => {
      let html = `<table><tr><th>Sediaan</th>`;
      Object.keys(frekMap).forEach(f => html += `<th>${f}</th>`);
      html += `</tr>`;
      
      sediaan.forEach(s => {
        html += `<tr><td><strong>${s.nama}</strong></td>`;
        Object.values(frekMap).forEach(calc => {
          const mgMin = calc.min, mgMax = calc.max;
          const mlMin = mgMin / s.mgPerMl, mlMax = mgMax / s.mgPerMl;
          html += `<td><span class="dosis-min">${fmt(mgMin)}</span>–<span class="dosis-max">${fmt(mgMax)}</span> mg<br><small>(${fmt(mlMin)}–${fmt(mlMax)} mL)</small></td>`;
        });
        html += `</tr>`;
      });
      html += `</table>`;
      return html;
    };

    function hitungSemua() {
      const BB = parseFloat(document.getElementById("berat").value);
      const show = (id, html, header) => {
        const el = document.getElementById(id);
        const h = document.getElementById("h" + id.replace("r", ""));
        if (BB > 0 && BB <= 100) {
          h.innerHTML = header;
          el.innerHTML = html;
          el.classList.add("active");
        } else {
          h.innerHTML = "";
          el.classList.remove("active");
        }
      };

      // AMBROXOL
      if (BB > 0) {
        const min = BB * 1.2, max = BB * 1.6;
        const header = `Dosis harian: <span class="dosis-min">${fmt(min)}</span> – <span class="dosis-max">${fmt(max)}</span> mg`;
        const html = buatTabel(
          [{nama:"Sirup 15 mg/5 mL", mgPerMl:3}, {nama:"Sirup 30 mg/5 mL", mgPerMl:6}, {nama:"Drops 15 mg/mL", mgPerMl:15}],
          {
            "2×/hari": {min: min/2, max: max/2},
            "3×/hari": {min: min/3, max: max/3}
          }
        );
        show("rAmbroxol", html, header);
      }

      // DOMPERIDON
      if (BB > 0) {
        const min = BB * 0.2, max = BB * 0.4, maxT = BB * 0.6;
        const header = `Dosis/kali: <span class="dosis-min">${fmt(min)}</span>–<span class="dosis-max">${fmt(max)}</span> mg <small>(opsional ${fmt(maxT)} mg)</small>`;
        const html = buatTabel(
          [{nama:"Sirup 5 mg/5 mL", mgPerMl:1}, {nama:"Drops 10 mg/mL", mgPerMl:10}],
          {
            "2×/hari": {min: min, max: max},
            "3×/hari": {min: min, max: max},
            "4×/hari": {min: min, max: max}
          }
        );
        show("rDomperidon", html, header + `<br><small style="color:#666">Maks harian: ${fmt(Math.min(BB*2.4,30))} mg</small>`);
      }

      // PARACETAMOL
      if (BB > 0) {
        const min = BB * 10, max = BB * 15;
        const header = `Dosis/kali: <span class="dosis-min">${fmt(min)}</span>–<span class="dosis-max">${fmt(max)}</span> mg`;
        const html = buatTabel(
          [{nama:"Drops 100 mg/mL", mgPerMl:100}, {nama:"Sirup 120 mg/5 mL", mgPerMl:24}],
          {
            "3×/hari": {min: min, max: max},
            "4×/hari": {min: min, max: max}
          }
        );
        show("rParacetamol", html, header + `<br><small style="color:#666">Maks 4×/hari (${fmt(BB*60)} mg/hari)</small>`);
      }

      // IBUPROFEN
      if (BB > 0) {
        const min = BB * 5, max = BB * 10;
        const header = `Dosis/kali: <span class="dosis-min">${fmt(min)}</span>–<span class="dosis-max">${fmt(max)}</span> mg`;
        const html = buatTabel(
          [{nama:"Sirup 100 mg/5 mL", mgPerMl:20}],
          {
            "3×/hari": {min: min, max: max},
            "4×/hari": {min: min, max: max}
          }
        );
        show("rIbuprofen", html, header + `<br><small style="color:#666">Maks 4×/hari (${fmt(BB*40)} mg/hari)</small>`);
      }

      // AMOXICILLIN
      if (BB > 0) {
        const min = BB * 25, max = BB * 50;
        const header = `Dosis harian: <span class="dosis-min">${fmt(min)}</span>–<span class="dosis-max">${fmt(max)}</span> mg`;
        const html = buatTabel(
          [{nama:"Sirup 125 mg/5 mL", mgPerMl:25}, {nama:"Sirup 250 mg/5 mL", mgPerMl:50}],
          {
            "2×/hari": {min: min/2, max: max/2},
            "3×/hari": {min: min/3, max: max/3}
          }
        );
        show("rAmoxicillin", html, header);
      }

      // CETIRIZINE
      if (BB > 0) {
        let min = Math.min(BB * 0.2, 10), max = Math.min(BB * 0.25, 10);
        const header = `Dosis sekali sehari: <span class="dosis-min">${fmt(min)}</span>–<span class="dosis-max">${fmt(max)}</span> mg <small>(maks 10 mg)</small>`;
        const html = buatTabel(
          [{nama:"Sirup 5 mg/5 mL", mgPerMl:1}, {nama:"Drops 10 mg/mL", mgPerMl:10}],
          { "1×/hari": {min: min, max: max} }
        );
        show("rCetirizine", html, header);
      }
    }

    document.addEventListener("DOMContentLoaded", hitungSemua);
  </script>
